1. Executive Summary & Overview

The AI-Based Fake Identity Document Screening System is a full-stack web application designed to automatically inspect Indian identity documents and determine their authenticity using client-side Computer Vision models and server-side Large Language Model (LLM) reasoning.

When an uploaded document is flagged as FAKE, an integrated AI assistant powered by Google Gemini provides real-time explanations detailing common visual, structural, and cryptographic red flags specific to that document type.

3. System Architecture & Workflow

System Execution Flow
  1. Authentication: User registers/logs in via login.html (session persisted in sessionStorage).
  2. Model Preloading: script.js preloads all 4 Teachable Machine TensorFlow.js models into browser GPU/memory upon startup.
  3. Classification: User selects document type and uploads an image. predictImage() runs instant WebGL inference.
  4. Result Rendering: Displays ORIGINAL or FAKE status with exact confidence percentage.
  5. AI Explanation Proxy: If flagged as FAKE, frontend sends a POST request to http://localhost:5001/api/explain-fake.
  6. Gemini Explanation: Express backend securely queries Google Gemini using persona system instructions and returns structured red-flag explanations.

// login.js - Key Session Check & Authentication
function checkAuth() {
    const isLoggedIn = sessionStorage.getItem('isLoggedIn');
    if (!isLoggedIn && window.location.pathname.includes('index.html')) {
        window.location.href = 'login.html';
    }
}

function loginUser(username, password) {
    const usersDB = JSON.parse(localStorage.getItem('usersDB') || '{}');
    if (usersDB[username] && usersDB[username].password === hashPassword(password)) {
        sessionStorage.setItem('isLoggedIn', 'true');
        window.location.href = 'index.html';
    } else {
        showError("Invalid username or password");
    }
}

// script.js - Centralized Document Registry & Concurrent Preloading
const DOC_TYPES = {
    aadhaar:  { name: "Aadhaar Card",  modelPath: "./model-aadhaar/" },
    pancard:  { name: "PAN Card",      modelPath: "./model-pancard/" },
    passport: { name: "Passport",      modelPath: "./model-passport/" },
    voterid:  { name: "Voter ID Card", modelPath: "./model-voterid/" }
};

let loadedModels = {};

async function preloadAllModels() {
    const loadPromises = Object.keys(DOC_TYPES).map(async (docKey) => {
        const modelURL = DOC_TYPES[docKey].modelPath + "model.json";
        const metadataURL = DOC_TYPES[docKey].modelPath + "metadata.json";
        loadedModels[docKey] = await tmImage.load(modelURL, metadataURL);
    });
    await Promise.all(loadPromises);
}

// Normalizes dataset class names (e.g. "ORIGINAL VOTEID" -> "ORIGINAL")
function normalizeClassName(className) {
    const upper = className.toUpperCase();
    if (upper.includes("ORIGINAL")) return "ORIGINAL";
    if (upper.includes("FAKE")) return "FAKE";
    return className;
}

app.post('/api/explain-fake', async (req, res) => {
    try {
        const { docType, confidence } = req.body;
        const prompt = `A ${docType} was classified as FAKE with ${confidence * 100}% confidence. Explain common visual anomalies.`;
